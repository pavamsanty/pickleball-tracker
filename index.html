<script>

  let players = [

    { name: 'Santosh', plus: 0 },

    { name: 'Alex', plus: 1 },

    { name: 'Dave', plus: 0 },

    { name: 'Elena', plus: 1 }

  ];


  window.onload = function() {

    document.getElementById('sessionDate').value = new Date().toISOString().split('T')[0];

    document.getElementById('courtCost').value = 60.00;

    renderPlayers();

  };


  function renderPlayers() {

    const list = document.getElementById('playerList');

    list.innerHTML = '';

    players.forEach((p, index) => {

      const row = document.createElement('div');

      row.className = 'player-row';

      row.innerHTML = `

        <input type="text" placeholder="Player Name" value="${p.name}" oninput="updateName(${index}, this.value)" />

        <input type="number" min="0" placeholder="+ Guests" value="${p.plus}" oninput="updatePlus(${index}, this.value)" title="Plus ones" />

        <button class="btn-danger" onclick="removePlayer(${index})">✕</button>

      `;

      list.appendChild(row);

    });

    updatePayerDropdown();

  }


  function addPlayerRow() {

    players.push({ name: '', plus: 0 });

    renderPlayers();

  }


  function removePlayer(index) {

    if (players.length > 1) {

      players.splice(index, 1);

      renderPlayers();

    }

  }


  function updateName(index, val) {

    players[index].name = val;

    updatePayerDropdown();

  }


  function updatePlus(index, val) {

    players[index].plus = parseInt(val, 10) || 0;

  }


  function updatePayerDropdown() {

    const select = document.getElementById('payerSelect');

    const prevSelected = select.value;

    select.innerHTML = '';

    players.forEach((p) => {

      const name = p.name.trim() || 'Unnamed';

      const opt = document.createElement('option');

      opt.value = name;

      opt.innerText = name;

      select.appendChild(opt);

    });

    if (prevSelected) select.value = prevSelected;

  }


  function calculateSplit() {

    const cost = parseFloat(document.getElementById('courtCost').value) || 0;

    const payerName = document.getElementById('payerSelect').value;


    let totalHeadcount = 0;

    players.forEach(p => {

      totalHeadcount += (1 + (parseInt(p.plus, 10) || 0));

    });


    if (totalHeadcount === 0 || cost <= 0) {

      alert('Please enter a valid court cost and at least one player.');

      return;

    }


    const costPerPerson = cost / totalHeadcount;

    const resultsCard = document.getElementById('resultsCard');

    const summaryInfo = document.getElementById('summaryInfo');

    const settlementList = document.getElementById('settlementList');


    summaryInfo.innerHTML = `

      <strong>Total Spots:</strong> ${totalHeadcount} &nbsp;|&nbsp; 

      <strong>Cost Per Person:</strong> $${costPerPerson.toFixed(2)}<br/>

      <strong>Paid Upfront By:</strong> ${payerName} ($${cost.toFixed(2)})

    `;


    settlementList.innerHTML = '';

    let hasDebts = false;


    players.forEach(p => {

      const name = p.name.trim() || 'Unnamed';

      if (name !== payerName) {

        const spots = 1 + (parseInt(p.plus, 10) || 0);

        const amountOwed = spots * costPerPerson;

        if (amountOwed > 0) {

          hasDebts = true;

          const div = document.createElement('div');

          div.className = 'settlement-item';

          div.innerHTML = `💸 <strong>${name}</strong> owes <strong>${payerName}</strong>: $${amountOwed.toFixed(2)} <span style="color:var(--text-muted); font-size:0.85rem">(${spots} spot${spots > 1 ? 's' : ''})</span>`;

          settlementList.appendChild(div);

        }

      }

    });


    if (!hasDebts) {

      settlementList.innerHTML = '<div class="settlement-item">All settled up! No payments pending.</div>';

    }


    resultsCard.style.display = 'block';

  }

</script>
